## 文件预览-说明文档

### doc-preview配置项

```js
options: {
    className: string = "docx", 
    // class name/prefix for default and document style classes
    // 用于默认样式和文档样式的类名或前缀

    inWrapper: boolean = true, 
    // enables rendering of wrapper around document content
    // 是否在文档内容外部包裹一个容器（wrapper）

    hideWrapperOnPrint: boolean = false, 
    // disable wrapper styles on print
    // 打印时是否隐藏 wrapper 的样式

    ignoreWidth: boolean = false, 
    // disables rendering width of page
    // 禁用页面宽度的渲染

    ignoreHeight: boolean = false, 
    // disables rendering height of page
    // 禁用页面高度的渲染

    ignoreFonts: boolean = false, 
    // disables fonts rendering
    // 禁用字体样式的渲染

    breakPages: boolean = true, 
    // enables page breaking on page breaks
    // 在分页符处启用分页

    ignoreLastRenderedPageBreak: boolean = true, 
    // disables page breaking on lastRenderedPageBreak elements
    // 忽略“lastRenderedPageBreak”元素的分页行为

    experimental: boolean = false, 
    // enables experimental features (tab stops calculation)
    // 启用实验性功能（如制表位计算等）

    trimXmlDeclaration: boolean = true, 
    // if true, xml declaration will be removed from xml documents before parsing
    // 是否在解析前移除 XML 文档的声明部分（<?xml...?>）

    useBase64URL: boolean = false, 
    // if true, images, fonts, etc. will be converted to base 64 URL, otherwise URL.createObjectURL is used
    // 是否将图片、字体等资源转换为 base64 URL，否则使用 URL.createObjectURL

    renderChanges: false, 
    // enables experimental rendering of document changes (insertions/deletions)
    // 启用对文档修改内容（插入/删除）的实验性渲染

    renderHeaders: true, 
    // enables headers rendering
    // 启用页眉渲染

    renderFooters: true, 
    // enables footers rendering
    // 启用页脚渲染

    renderFootnotes: true, 
    // enables footnotes rendering
    // 启用脚注渲染

    renderEndnotes: true, 
    // enables endnotes rendering
    // 启用尾注渲染

    renderComments: false, 
    // enables experimental comments rendering
    // 启用实验性评论（批注）渲染

    renderAltChunks: true, 
    // enables altChunks (html parts) rendering
    // 启用 altChunk（HTML 片段）内容的渲染

    debug: boolean = false, 
    // enables additional logging
    // 启用额外的日志输出，便于调试
}
```
